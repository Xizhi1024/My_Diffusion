```python
class InputLayer:
    """
    DICOM → 多窗2.5D张量
    """
    ct_window_configs = {
        "soft_tissue": (50, 350),    # 窗宽/窗位: 软组织窗
        "bone":        (1500, 400),  # 骨窗
        "lung":        (1500, -600), # 肺窗
        "wide":        (400, 40),    # 宽窗
    }
    num_slices = 5  # z-2 : z+2, 2.5D堆叠

    # 输出 shape: [B, 4_windows * 5_slices = 20, H, W]

    def dcm_to_hu(ct_dicom):
        """DICOM → HU值 (RescaleSlope * pixel + Intercept)"""
        return slope * pixel_array + intercept

    def hu_to_window(hu, width, level):
        """HU值 → 窗宽窗位映射 → [-1, 1]"""
        lower = level - width / 2
        upper = level + width / 2
        return 2 * (hu.clamp(lower, upper) - lower) / width - 1

    def forward(ct_dicom_5slices):
        hu_stack = [self.dcm_to_hu(s) for s in ct_dicom_5slices]
        return concat([
            self.hu_to_window(hu_stack, **w)
            for w in window_configs.values()
        ], dim=1)  # [B, 20, H, W]
```

**关键决策**：PNG 仅用于可视化和数据检查，模型输入必须走 DICOM→HU→NIfTI 管线，保物理语义。

```python
class ControlNetInjection(nn.Module):
    """
    文档4.1: 锁定扩散U-Net骨干，可训练copy注入多尺度条件
    """
    def __init__(self, unet_encoder):
        # 复制U-Net encoder的每层结构 (参数独立、可训练)
        self.trainable_copy = copy_structure(unet_encoder)
        # 零卷积层: 初始化weight=0, bias=0 → 训练起始不破坏预训练分布
        self.zero_convs = nn.ModuleList([
            zero_module(nn.Conv2d(c, c, 1)) for c in encoder_channels
        ])
    
    def forward(self, ct_features, timestep):
        encoded = self.trainable_copy(ct_features)
        return [zero_conv(f) for f, zero_conv in zip(encoded, self.zero_convs)]
        # 输出 → 加到U-Net skip connections

class CrossAttentionBottleneck(nn.Module):
    """
    文档4.2: U-Net bottleneck处cross-attention
    条件token [GNN嵌入 + DINOv2特征 + Hotspot pooled] → query by UNet features
    """
    def __init__(self, unet_dim, cond_dim, num_heads=8):
        self.cross_attn = nn.MultiheadAttention(unet_dim, num_heads, batch_first=True)
        self.cond_proj = nn.Linear(cond_dim, unet_dim)
    
    def forward(self, unet_feat, condition_tokens):
        # unet_feat: [B, H*W, D_unet]
        # condition_tokens: [B, N_tokens, D_cond]
        k = v = self.cond_proj(condition_tokens)
        q = unet_feat
        attended, _ = self.cross_attn(q, k, v)
        return attended + unet_feat  # residual
```

#### Mamba Bottleneck（文档5.2）

```python
class MambaBlock(nn.Module):
    """
    选择性状态空间块: O(N)复杂度替代O(N²)自注意力
    """
    def __init__(self, dim, d_state=16, d_conv=4, expand=2):
        self.norm = nn.LayerNorm(dim)
        self.ssm = Mamba(
            d_model=dim,
            d_state=d_state,        # SSM state dimension
            d_conv=d_conv,          # 局部卷积核
            expand=expand,          # 内部扩展因子
        )
    
    def forward(self, x):
        # x: [B, L, D]  (L = H/32 * W/32)
        residual = x
        x = self.norm(x)
        x = self.ssm(x)
        return x + residual

class MambaBottleneck(nn.Module):
    def __init__(self, dim, num_layers=6):
        self.layers = nn.ModuleList([
            MambaBlock(dim) for _ in range(num_layers)
        ])
    def forward(self, x):
        for layer in self.layers:
            x = layer(x)
        return x
```

```python
    文档6.1: CT→PET布朗桥扩散

    """
    def __init__(self, T=1000):

class ScaleAwareNoiseScheduler(nn.Module):
    """
    文档6.4: 尺度感知噪声调度
    不同空间频率尺度使用不同噪声注入速率
    小目标高频信号在forward过程中被保护更久
    """
    def __init__(self, T=1000, num_scales=3):
        # 低频 → 标准噪声速率 (快速破坏全局结构)
        # 中频 → 中等速率
        # 高频 → 慢速率 (保护小病灶边界)
        self.noise_rates = nn.Parameter(torch.ones(num_scales, T))
        
    def decompose_and_noise(self, x, t):
        """拉普拉斯金字塔分解 → 每层不同噪声 → 重建noisy x_t"""
        levels = self.laplacian_pyramid(x)  # [低频, 中频, 高频残差]
        noisy_levels = []
        for i, level in enumerate(levels):
            noise_rate = self.noise_rates[i, t]
            noisy_levels.append(
                level * (1 - noise_rate).sqrt() + 
                torch.randn_like(level) * noise_rate.sqrt()
            )
        return self.reconstruct(noisy_levels)
```

## 三、Loss 计算层

```python
class TotalLoss(nn.Module):
    """
    文档第七章: 分层组合Loss, 按Phase叠加
    """
    def forward(self, pred_pet, pred_logvar, true_pet, pred_hotspot, 
                true_hotspot, seg_pred, seg_true, ct_features, pet_features):
        loss = 0
        
        # === Phase 1: 基础Loss ===
        loss += self.L1(pred_pet, true_pet)                    # 7.1.1
        loss += self.MSE(pred_pet, true_pet)                   # 7.1.1
        loss += self.gradient_loss(pred_pet, true_pet)         # 7.2.1
        
        # === Phase 2: 高ROI Loss ===
        loss += self.topk_focal_loss(pred_pet, true_pet,       # 7.3.1
                                      k=0.05, gamma=2.0)
        loss += self.focal_frequency_loss(pred_pet, true_pet,  # 7.2.2
                                           alpha=1.0)
        loss += self.patch_nce_loss(ct_features, pet_features, # 7.2.3
                                     temperature=0.07)
        
        # === Phase 6: 闭环+可靠性 ===
        loss += self.seg_consistency_loss(seg_pred, seg_true)  # 7.4.2
        loss += self.heteroscedastic_nll(pred_pet, pred_logvar,# 7.5.1
                                          true_pet)
        
        # === 辅助Loss (可选) ===
        loss += self.roi_suv_loss(pred_pet, true_pet,           # 7.3.2
                                   lesion_mask)
        loss += self.sinkhorn_loss(pred_pet, true_pet)          # 7.1.2
        
        return loss

    def topk_focal_loss(self, pred, true, k=0.05, gamma=2.0):
        """
        文档7.3.1: 只关注PET最亮的前k%像素
        """
        threshold = torch.quantile(true, 1 - k)
        mask = (true >= threshold).float()
        abs_error = (pred - true).abs()
        focal_weight = (1 - torch.exp(-abs_error)) ** gamma
        return (mask * focal_weight * abs_error).sum() / (mask.sum() + 1e-8)

    def patch_nce_loss(self, ct_feat, pet_feat, temperature=0.07):
        """
        文档7.2.3: 每个空间位置CT特征只与同位置PET特征相似
        """
        B, C, H, W = ct_feat.shape
        ct_flat = ct_feat.reshape(B, C, -1).permute(0, 2, 1)  # [B, H*W, C]
        pet_flat = pet_feat.reshape(B, C, -1).permute(0, 2, 1)
        
        # 正样本: 同位置 (ct_i, pet_i)
        pos_sim = (ct_flat * pet_flat).sum(-1) / temperature  # [B, H*W]
        
        # 负样本: 不同位置 (ct_i, pet_j), j≠i
        neg_sim = torch.bmm(ct_flat, pet_flat.transpose(1, 2)) / temperature
        
        # InfoNCE loss
        numerator = pos_sim.exp()
        denominator = neg_sim.exp().sum(-1)
        return -(numerator / denominator).log().mean()
```

```
  + Top-K Focal Loss
  + Focal Frequency Loss
  + PatchNCE Loss
```

### Phase 3: Encoder升级

```
CT DICOM ──┬── Multi-Scale Stem + DualStream ──┐
            │                                      ├── [c1,c2,c3,c4]
            └── Frequency Branch (Gabor) ───┘
                                             Mamba替代CNN Encoder
```

### Phase 4: 先验注入

```
CT DICOM ──┬── Encoder (Phase3版)
            │      │
            ├── HotspotPriorNet ── soft attn map ──→ concat to ControlNet
            │                                       │
            └── 器官分割 → GNN ──→ token ──→ Cross-Attn Bottleneck
                                                  │
元数据(年龄/血糖) → MLP ──→ FiLM ──→ U-Net各层
```

### Phase 5: 扩散升级

```
标准DDPM → 布朗桥扩散 (BBDM)
         → 尺度感知噪声调度
         → Latent Diffusion (VAE编解码)
         → DDIM 20步采样
```

### Phase 6: 闭环+可靠性

```
合成PET → 冻结分割器 → seg loss梯度反传 → 生成器
合成PET → 异方差NLL → 不确定性图 联合输出
```

## 五、关键设计原则

| 原则 | 说明 |
|------|------|
| **分层注入，不混用** | 空间走ControlNet、语义走Cross-Attn、全局走FiLM——不搞大一统融合 |
| **物理语义不丢失** | CT保留HU、PET保留SUV，不在归一化中丢失临床意义 |
| **先验互补，不重叠** | 频域Gabor(纹理) + Hotspot(代谢) + GNN(解剖) 各司其职 |
| **损失按层叠加** | Phase2先加3个高ROI Loss验证效果，不要一次性加全部 |
| **Phase隔离推进** | 每阶段独立评估，确认增益后再进入下一Phase |
